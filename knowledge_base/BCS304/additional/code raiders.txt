1.Develop a Program in C for the following:
a) Declare a calendar as an array of 7 elements (A dynamically Created array) to represent
7 days of a week. Each Element of the array is a structure having three fields. The first
field is the name of the Day (A dynamically allocated String), The second field is the
date of the Day (A integer), the third field is the description of the activity for a
particular day (A dynamically allocated String).
b) Write functions create(), read() and display(); to create the calendar, to read the data
from the keyboard and to print weeks activity details report on screen.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DAYS 7
#define MAX_LEN 50

// Structure for a day
typedef struct {
    char *dayName;    // dynamically allocated string
    int date;
    char *activity;   // dynamically allocated string
} Calendar;

// Function to create calendar dynamically
Calendar* create() {
    Calendar *week = (Calendar*)malloc(DAYS * sizeof(Calendar));
    if (week == NULL) {
        printf("Memory not allocated!\n");
        exit(0);
    }

    // Allocate memory for day names and activities
    for (int i = 0; i < DAYS; i++) {
        week[i].dayName = (char*)malloc(MAX_LEN * sizeof(char));
        week[i].activity = (char*)malloc(MAX_LEN * sizeof(char));
        if (!week[i].dayName || !week[i].activity) {
            printf("Memory allocation failed!\n");
            exit(0);
        }
    }

    return week;
}

// Function to read data from user
void read(Calendar *week) {
    printf("Enter details for 7 days:\n");
    for (int i = 0; i < DAYS; i++) {
        printf("\nDay %d Name: ", i + 1);
        scanf(" %[^\n]", week[i].dayName);  // read string with spaces
        printf("Date (integer): ");
        scanf("%d", &week[i].date);
        printf("Activity Description: ");
        scanf(" %[^\n]", week[i].activity);
    }
}

// Function to display calendar
void display(Calendar *week) {
    printf("\n------ WEEKLY ACTIVITY REPORT ------\n");
    printf("%-10s %-5s %-30s\n", "Day", "Date", "Activity");
    printf("-------------------------------------------\n");
    for (int i = 0; i < DAYS; i++) {
        printf("%-10s %-5d %-30s\n", week[i].dayName, week[i].date, week[i].activity);
    }
}

int main() {
    Calendar *week;

    week = create();
    read(week);
    display(week);

    // Free allocated memory
    for (int i = 0; i < DAYS; i++) {
        free(week[i].dayName);
        free(week[i].activity);
    }
    free(week);

    return 0;
}

: 2.Develop a Program in C for the following operations on Strings.
a. Read a main String (STR), a Pattern String (PAT) and a Replace String (REP)
b. Perform Pattern Matching Operation: Find and Replace all occurrences of PAT in
STR with REP if PAT exists in STR. Report suitable messages in case PAT does not
exist in STR
Support the program with functions for each of the above operations. Don't use Built-in
functions. 

#include <stdio.h>
#include <stdlib.h>

#define MAX 1000

// Function to read strings
void readStrings(char *STR, char *PAT, char *REP) {
    printf("Enter main string (STR): ");
    scanf(" %[^\n]", STR);
    printf("Enter pattern string (PAT): ");
    scanf(" %[^\n]", PAT);
    printf("Enter replace string (REP): ");
    scanf(" %[^\n]", REP);
}

// Function to find length of a string
int strLen(char *s) {
    int len = 0;
    while (s[len] != '\0') len++;
    return len;
}

// Function to check if PAT occurs at STR[pos]
int matchAt(char *STR, char *PAT, int pos) {
    int i = 0;
    while (PAT[i] != '\0') {
        if (STR[pos + i] == '\0' || STR[pos + i] != PAT[i])
            return 0; // mismatch
        i++;
    }
    return 1; // match
}

// Function to replace all occurrences of PAT in STR with REP
void replacePattern(char *STR, char *PAT, char *REP) {
    int lenSTR = strLen(STR);
    int lenPAT = strLen(PAT);
    int lenREP = strLen(REP);

    char result[MAX];
    int i = 0, j = 0;
    int found = 0;

    while (i < lenSTR) {
        if (matchAt(STR, PAT, i)) {
            found = 1;
            // Copy REP into result
            for (int k = 0; k < lenREP; k++)
                result[j++] = REP[k];
            i += lenPAT; // skip PAT in STR
        } else {
            result[j++] = STR[i++];
        }
    }

    result[j] = '\0';

    if (found) {
        printf("\nPattern replaced successfully!\n");
        printf("Updated string: %s\n", result);
    } else {
        printf("\nPattern '%s' not found in the main string.\n", PAT);
    }
}

int main() {
    char STR[MAX], PAT[MAX], REP[MAX];

    readStrings(STR, PAT, REP);
    replacePattern(STR, PAT, REP);

    return 0;
}

 3.Develop a menu driven Program in C for the following operations on STACK of Integers
(Array Implementation of Stack with maximum size MAX)
a. Push an Element on to Stack
b. Pop an Element from Stack
c. Demonstrate how Stack can be used to check Palindrome
d. Demonstrate Overflow and Underflow situations on Stack
e. Display the status of Stack
f. Exit
Support the program with appropriate functions for each of the above operation.

#include <stdio.h>
#include <stdlib.h>

#define MAX 5   // Maximum size of the stack

int stack[MAX];
int top = -1;

// Function to push element onto stack
void push() {
    if (top == MAX - 1) {
        printf("Stack Overflow! Cannot push more elements.\n");
        return;
    }
    int x;
    printf("Enter element to push: ");
    scanf("%d", &x);
    stack[++top] = x;
    printf("%d pushed onto stack.\n", x);
}

// Function to pop element from stack
void pop() {
    if (top == -1) {
        printf("Stack Underflow! No element to pop.\n");
        return;
    }
    int x = stack[top--];
    printf("Popped element: %d\n", x);
}

// Function to display stack
void display() {
    if (top == -1) {
        printf("Stack is empty.\n");
        return;
    }
    printf("Stack elements (top to bottom): ");
    for (int i = top; i >= 0; i--)
        printf("%d ", stack[i]);
    printf("\n");
}

// Function to check if stack forms a palindrome
void checkPalindrome() {
    if (top == -1) {
        printf("Stack is empty. Cannot check palindrome.\n");
        return;
    }
    int i = 0, j = top;
    int isPalindrome = 1;
    while (i < j) {
        if (stack[i] != stack[j]) {
            isPalindrome = 0;
            break;
        }
        i++;
        j--;
    }
    if (isPalindrome)
        printf("Stack elements form a palindrome.\n");
    else
        printf("Stack elements do NOT form a palindrome.\n");
}

// Function to demonstrate overflow and underflow
void demoOverflowUnderflow() {
    printf("Demonstrating underflow:\n");
    pop();  // Attempt to pop (may underflow)

    printf("Demonstrating overflow:\n");
    for (int i = 1; i <= MAX + 2; i++) {
        printf("Pushing %d: ", i*10);
        if (top == MAX - 1)
            printf("Stack Overflow! Cannot push %d\n", i*10);
        else {
            stack[++top] = i*10;
            printf("Pushed %d\n", i*10);
        }
    }
}

int main() {
    int choice;
    do {
        printf("\n--- STACK MENU ---\n");
        printf("1. Push an Element\n");
        printf("2. Pop an Element\n");
        printf("3. Check Palindrome\n");
        printf("4. Demonstrate Overflow/Underflow\n");
        printf("5. Display Stack\n");
        printf("6. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice) {
            case 1: push(); break;
            case 2: pop(); break;
            case 3: checkPalindrome(); break;
            case 4: demoOverflowUnderflow(); break;
            case 5: display(); break;
            case 6: printf("Exiting...\n"); break;
            default: printf("Invalid choice!\n");
        }
    } while(choice != 6);

    return 0;
}

 4.Develop a Program in C for converting an Infix Expression to Postfix Expression. Program
should support for both parenthesized and free parenthesized
expressions with the operators: +, -, *, /, % (Remainder), ^ (Power) and alphanumeric
operands.
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

#define MAX 100

char stack[MAX];
int top = -1;

// Push function
void push(char c) {
    if (top == MAX - 1) {
        printf("Stack Overflow!\n");
        return;
    }
    stack[++top] = c;
}

// Pop function
char pop() {
    if (top == -1)
        return '\0';
    return stack[top--];
}

// Peek function
char peek() {
    if (top == -1)
        return '\0';
    return stack[top];
}

// Function to return precedence of operators
int precedence(char op) {
    switch(op) {
        case '^': return 3;
        case '*': case '/': case '%': return 2;
        case '+': case '-': return 1;
        default: return 0;
    }
}

// Function to check if character is operator
int isOperator(char c) {
    return (c == '+' || c == '-' || c == '*' || c == '/' || c == '%' || c == '^');
}

// Function to convert infix to postfix
void infixToPostfix(char infix[], char postfix[]) {
    int i = 0, j = 0;
    char c;

    while (infix[i] != '\0') {
        c = infix[i];

        if (isalnum(c)) {
            postfix[j++] = c;  // Operand goes directly to postfix
        }
        else if (c == '(') {
            push(c);
        }
        else if (c == ')') {
            while (top != -1 && peek() != '(')
                postfix[j++] = pop();
            pop(); // Remove '('
        }
        else if (isOperator(c)) {
            while (top != -1 && precedence(peek()) >= precedence(c)) {
                if(c == '^' && peek() == '^')  // Right-associative for ^
                    break;
                postfix[j++] = pop();
            }
            push(c);
        }
        i++;
    }

    while (top != -1)
        postfix[j++] = pop();

    postfix[j] = '\0';
}

int main() {
    char infix[MAX], postfix[MAX];

    printf("Enter an infix expression: ");
    scanf(" %[^\n]", infix);

    infixToPostfix(infix, postfix);

    printf("Postfix expression: %s\n", postfix);

    return 0;
}

 5.Develop a Program in C for the following Stack Applications
a. Evaluation of Suffix expression with single digit operands and operators: +, -, *, /, %,^
b. Solving Tower of Hanoi problem with n disks


#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <ctype.h>

#define MAX 100

// ------------------- STACK FOR POSTFIX EVALUATION -------------------
int stack[MAX];
int top = -1;

// Push function
void push(int x) {
    if (top == MAX - 1) {
        printf("Stack Overflow!\n");
        return;
    }
    stack[++top] = x;
}

// Pop function
int pop() {
    if (top == -1) {
        printf("Stack Underflow!\n");
        return -1;
    }
    return stack[top--];
}

// Evaluate single-digit postfix expression
void evaluatePostfix() {
    char expr[MAX];
    printf("Enter a postfix expression (single-digit operands): ");
    scanf(" %[^\n]", expr);

    int i = 0;
    while (expr[i] != '\0') {
        char c = expr[i];
        if (isdigit(c)) {
            push(c - '0');  // Convert char to int
        } else {
            int op2 = pop();
            int op1 = pop();
            int res;

            switch(c) {
                case '+': res = op1 + op2; break;
                case '-': res = op1 - op2; break;
                case '*': res = op1 * op2; break;
                case '/': res = op1 / op2; break;
                case '%': res = op1 % op2; break;
                case '^': res = (int)pow(op1, op2); break;
                default:
                    printf("Invalid operator %c\n", c);
                    return;
            }
            push(res);
        }
        i++;
    }

    printf("Result of postfix expression: %d\n", pop());
}

// ------------------- TOWER OF HANOI -------------------
void towerOfHanoi(int n, char from, char to, char aux) {
    if (n == 1) {
        printf("Move disk 1 from %c to %c\n", from, to);
        return;
    }
    towerOfHanoi(n - 1, from, aux, to);
    printf("Move disk %d from %c to %c\n", n, from, to);
    towerOfHanoi(n - 1, aux, to, from);
}

// ------------------- MAIN MENU -------------------
int main() {
    int choice, n;

    do {
        printf("\n--- STACK APPLICATION MENU ---\n");
        printf("1. Evaluate Postfix Expression\n");
        printf("2. Solve Tower of Hanoi\n");
        printf("3. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice) {
            case 1:
                top = -1; // Reset stack
                evaluatePostfix();
                break;
            case 2:
                printf("Enter number of disks: ");
                scanf("%d", &n);
                printf("Tower of Hanoi solution for %d disks:\n", n);
                towerOfHanoi(n, 'A', 'C', 'B');
                break;
            case 3:
                printf("Exiting...\n");
                break;
            default:
                printf("Invalid choice!\n");
        }
    } while(choice != 3);

    return 0;
}
